"""Exercise rejection of an actual API-compatible local ELF substituted into an independent package copy."""
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

repository = Path('/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom')
source = Path('/dev/shm/rustdom-numeric-hosted-package-20261026')
target = Path('/dev/shm/rustdom-numeric-local-negative-20261026')
assert not target.exists()
shutil.copytree(source, target, symlinks=True)
local_binary = Path('/dev/shm/rustdom-range-numeric-composed-v1/qualified-source-reviewed-node24/dist/rustdom.node')
original = (target / 'candidate/dist/rustdom.node').read_bytes()
wrong = local_binary.read_bytes()
assert hashlib.sha256(original).hexdigest() == 'a2777cbf00baa2cc7aa67c1098d6a5ceba4009aa6e514d558aac0b304e7ea583'
assert hashlib.sha256(wrong).hexdigest() == 'a6a21c3c97f6972eb871be739fd4e4f9d7d4030156ab8062c69bab5a6d80c3b0'
(target / 'candidate/dist/rustdom.node').write_bytes(wrong)
command = ['/mnt/c/Users/guido/ghq/projects/rustdom/.tools/node/bin/node', '--expose-gc',
    str(repository / 'reports/benchmarks/native-class-ownership/numeric-composed-packages/smoke.cjs'), str(target)]
result = subprocess.run(command, capture_output=True, timeout=180)
output = repository / 'reports/validation/range-numeric-composed-package/20261026'
(output / 'wrong-local-elf.stdout.log').write_bytes(result.stdout)
(output / 'wrong-local-elf.stderr.log').write_bytes(result.stderr)
record = {'argv': command, 'exitCode': result.returncode, 'hostedNativeSha256': hashlib.sha256(original).hexdigest(),
    'substitutedLocalSha256': hashlib.sha256(wrong).hexdigest(), 'candidateRuntimeLoaded': False,
    'runtimeLoadPreventedByReviewedPreAuthentication': True, 'benchmarkExecuted': False, 'performanceAccepted': None, 'gateReevaluated': False}
(output / 'wrong-local-elf.json').write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8', newline='\n')
assert result.returncode != 0 and result.stdout == b'' and b'candidate/dist/rustdom.node' in result.stderr
assert (source / 'candidate/dist/rustdom.node').read_bytes() == original
print(json.dumps({'wrongLocalElfRejected': True, 'originalPreserved': True, 'exitCode': result.returncode}))
