"""Exercise the exact proposed sparse hydration against a native Git fixture containing the current input bytes."""
import hashlib
import json
from pathlib import Path
import subprocess

repository = Path('/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom')
root = Path('/dev/shm/rustdom-numeric-ci-minimal-20261026')
metadata = Path('/tmp/rustdom-numeric-ci-fixture-git-20261026')
output = repository / 'reports/validation/numeric-package-ci/20261026'
receipt = json.loads((output / 'minimal-snapshot.json').read_bytes())
assert not (root / '.git').exists() and not metadata.exists()
for relative, expected in receipt['files'].items():
    raw = (root / relative).read_bytes()
    assert len(raw) == expected['bytes'] and hashlib.sha256(raw).hexdigest() == expected['sha256']
def git(*arguments, input_bytes=None):
    return subprocess.check_output(['git', '-c', 'core.longpaths=true', *arguments], cwd=root, input=input_bytes)
git('init', '--initial-branch=main', '--separate-git-dir=' + str(metadata))
git('config', 'core.autocrlf', 'false')
git('--literal-pathspecs', 'add', '--pathspec-from-file=-', '--pathspec-file-nul', input_bytes='\0'.join(receipt['files']).encode() + b'\0')
git('-c', 'user.name=Native CI fixture', '-c', 'user.email=fixture@example.invalid', 'commit', '-m', 'Native sparse preparation behavior fixture')
fixture_head = git('rev-parse', 'HEAD').decode().strip()
patterns = b'''/*
!/*/
/reports/benchmarks/native-class-ownership/numeric-composed-packages/
/reports/validation/range-numeric-composed-package/20261026/archive.json
/reports/validation/owner-batch-capture-method/review.json
'''
git('sparse-checkout', 'set', '--no-cone', '--stdin', input_bytes=patterns)
assert not (root / 'src/lib.rs').exists()
command = ['python3', '-B', str(root / 'reports/benchmarks/native-class-ownership/numeric-composed-packages/checkout.py'), str(root)]
result = subprocess.run(command, capture_output=True, timeout=180)
(output / 'native-sparse-hydration.stdout.json').write_bytes(result.stdout)
(output / 'native-sparse-hydration.stderr.log').write_bytes(result.stderr)
assert result.returncode == 0
prepared = json.loads(result.stdout)
assert prepared['head'] == fixture_head and prepared['requestedFiles'] == receipt['count']
for relative, expected in receipt['files'].items():
    raw = (root / relative).read_bytes()
    assert len(raw) == expected['bytes'] and hashlib.sha256(raw).hexdigest() == expected['sha256']
assert git('status', '--porcelain=v1') == b''
record = {'fixtureHead': fixture_head, 'fixtureGitMetadataIsNativeLinux': True,
    'fixtureIsPublishedMethodCommit': False, 'publishedSourceCommitImpersonated': False,
    'inputFiles': receipt['count'], 'allCopiedInputBytesRestored': True, 'exitCode': result.returncode,
    'runtimeExecuted': False, 'benchmarkExecuted': False}
(output / 'native-sparse-fixture.json').write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8', newline='\n')
print(json.dumps(record))
