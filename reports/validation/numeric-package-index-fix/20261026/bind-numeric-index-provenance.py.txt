import hashlib
import json
from pathlib import Path

repository = Path(__file__).resolve().parent.parent
root = repository / 'reports/benchmarks/native-class-ownership/numeric-composed-packages'
manifest = 'reports/validation/range-numeric-composed-package/20261026/archive.json'
checksum = hashlib.sha256((repository / manifest).read_bytes()).hexdigest()
assert checksum == '4d68ad3bd8cf5a997b04545376ceae703fa8df8e185797a6b47e11f4de0e9d99'
policy_path = root / 'constants/package.json'
policy = json.loads(policy_path.read_bytes())
assert 'publishedArchiveManifest' not in policy
policy['publishedArchiveManifest'] = {'path': manifest, 'sha256': checksum}
policy_path.write_text(json.dumps(policy, indent=2) + '\n', encoding='utf-8', newline='\n')
assembler_sha = hashlib.sha256((root / 'assemble.py').read_bytes()).hexdigest()
policy_sha = hashlib.sha256(policy_path.read_bytes()).hexdigest()
smoke_path = root / 'constants/smoke.json'; smoke = json.loads(smoke_path.read_bytes())
smoke.update(assemblerSha256=assembler_sha, assemblyPolicySha256=policy_sha)
smoke_path.write_text(json.dumps(smoke, indent=2) + '\n', encoding='utf-8', newline='\n')
transport_path = root / 'constants/transport.json'; transport = json.loads(transport_path.read_bytes())
transport['assembler']['sha256'] = assembler_sha
transport['assemblyPolicy']['sha256'] = policy_sha
transport_path.write_text(json.dumps(transport, indent=2) + '\n', encoding='utf-8', newline='\n')
print(json.dumps({'assemblerSha256': assembler_sha, 'policySha256': policy_sha, 'archiveManifestSha256': checksum}))
