import hashlib
import json
import pathlib
import shutil
import subprocess
import sys
import tarfile
if sys.flags.optimize:raise RuntimeError('Preservation requires non-optimized Python')
root=pathlib.Path('/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom')
source=pathlib.Path('/run/rustdom-source-profile-qualification-20261027')
output=root/'reports/validation/block-confirmation-source-profile/20261027';output.mkdir(exist_ok=False,parents=True)
members={member.relative_to(source).as_posix():dict(bytes=member.stat().st_size,sha256=hashlib.sha256(member.read_bytes()).hexdigest()) for member in source.rglob('*') if member.is_file()}
archive=output/'qualification.tar.gz'
with tarfile.open(archive,'w:gz') as bundle:
    for name in sorted(members):bundle.add(source/name,arcname=name,recursive=False)
with tarfile.open(archive,'r:gz') as bundle:
    assert len(bundle.getmembers())==len(members)
    for member in bundle.getmembers():
        assert member.isfile() and member.name in members
        raw=bundle.extractfile(member).read();assert len(raw)==members[member.name]['bytes']
        assert hashlib.sha256(raw).hexdigest()==members[member.name]['sha256']
(output/'manifest.json').write_text(json.dumps(members,indent=2)+'\n')
cache=root/'.cache/confirmation-authentication-20261027'
