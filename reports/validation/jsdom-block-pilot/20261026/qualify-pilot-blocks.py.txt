import hashlib
import json
from pathlib import Path
import subprocess

repository = Path(__file__).resolve().parent.parent
primary = Path(r'C:\Users\guido\ghq\projects\rustdom')
output = repository / 'reports/validation/jsdom-block-pilot/20261026'
output.mkdir(parents=True, exist_ok=True)
test_path = 'reports/benchmarks/native-class-ownership/control-precision/pilot/blocks.spec.cjs'
versions = {
    '22.12.0': 'b3b117a08ee61efee09e6fd523ab33c0c018da1b570bde08e4fd914dc1170ed6',
    '24.14.1': '58e74bf02fc5bbacc41dcb8bef089961cd5bddd37830b87784e4fc624d145d1f'}
results = []
for version, expected in versions.items():
    executable = primary / f'.tools/node-win/node-v{version}-win-x64/node.exe'
    assert hashlib.sha256(executable.read_bytes()).hexdigest() == expected
    actual = subprocess.check_output([str(executable), '--version']).decode().strip()
    assert actual == 'v' + version
    result = subprocess.run([str(executable), '--test', test_path], cwd=repository, capture_output=True)
    for name, contents in [('stdout', result.stdout), ('stderr', result.stderr)]:
        with (output / f'node-{version}-windows.{name}.log').open('xb') as stream: stream.write(contents)
    results.append({'node': actual, 'executableSha256': expected, 'platform': 'win32', 'exitCode': result.returncode})
with (output / 'qa.json').open('x', encoding='utf-8') as stream:
    json.dump({'results': results, 'syntheticScalarContractsOnly': True,
        'domExecuted': False, 'scientificCaptureExecuted': False,
        'performanceAccepted': None, 'gateReevaluated': False}, stream, indent=2)
print(json.dumps(results))
assert all(result['exitCode'] == 0 for result in results)
