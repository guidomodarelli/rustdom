import hashlib
import json
import shutil
import subprocess
repository=Path('/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom')
primary=Path('/mnt/c/Users/guido/ghq/projects/rustdom')
source=Path('/dev/shm/rustdom-pilot-lifecycle-fix-source-20261026')
output=repository/'.cache/pilot-lifecycle-zombie-fix-20261026'
scratch=Path('/dev/shm/rustdom-pilot-lifecycle-zombie-fix-20261026')
assert not source.exists() and not output.exists() and not scratch.exists()
output.mkdir();scratch.mkdir()
