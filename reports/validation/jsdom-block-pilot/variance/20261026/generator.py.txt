"""Generate a numerical reference from high-precision incomplete-gamma inversion; no DOM or timing inputs."""
import hashlib
import importlib.metadata
import json
from pathlib import Path
import sys

repository = Path(__file__).resolve().parent.parent
sys.path.insert(0, r'C:\Users\guido\ghq\projects\rustdom\.tools\statistical-reference')
import mpmath as mathematics
assert mathematics.__version__ == '1.3.0'
mathematics.mp.dps = 80
degrees = 15

def quantile(probability):
    lower = mathematics.mpf(0); upper = mathematics.mpf(64)
    for unused in range(300):
        middle = (lower + upper) / 2
        cdf = mathematics.gammainc(mathematics.mpf(degrees) / 2, 0, middle / 2, regularized=True)
        if cdf < probability: lower = middle
        else: upper = middle
    value = (lower + upper) / 2
    residual = mathematics.gammainc(mathematics.mpf(degrees) / 2, 0, value / 2, regularized=True) - probability
    assert abs(residual) < mathematics.mpf('1e-70')
    return value, residual

probabilities = ['0.025', '0.05', '0.975']
quantiles = {probability: quantile(mathematics.mpf(probability)) for probability in probabilities}
directory = repository / 'reports/benchmarks/native-class-ownership/control-precision/pilot/constants'
policy = {'degreesOfFreedom': degrees, 'blocks': 16, 'confidenceLevel': 0.95,
    'lowerTailQuantile025': float(quantiles['0.025'][0]), 'lowerTailQuantile05': float(quantiles['0.05'][0]),
    'upperTailQuantile975': float(quantiles['0.975'][0]), 'source': 'mpmath-1.3.0-incomplete-gamma-bisection-80-decimal-digits',
    'model': 'iid-normal-block-contrasts-pointwise-only',
    'sources': ['https://www.itl.nist.gov/div898/handbook/eda/section3/eda358.htm', 'https://www.itl.nist.gov/div898/handbook/eda/section3/eda3674.htm']}
(directory / 'variance.json').write_bytes(json.dumps(policy, indent=2).encode() + b'\n')
blocks = [{'baselineMeanMs': 100 + 2 * index, 'candidateMeanMs': 101 + 3 * index + 2 * (index % 3 - 1)} for index in range(16)]
expected = []
for ratio in ['0.98', '1', '1.02']:
    values = [mathematics.mpf(block['candidateMeanMs']) - mathematics.mpf(ratio) * block['baselineMeanMs'] for block in blocks]
    mean = sum(values) / 16; variance = sum((value - mean) ** 2 for value in values) / degrees; deviation = mathematics.sqrt(variance)
    expected.append({'fixedRatio': float(ratio), 'meanMs': float(mean), 'sampleVarianceMsSquared': float(variance),
        'sampleStandardDeviationMs': float(deviation), 'standardDeviationIntervalMs': [
            float(deviation * mathematics.sqrt(degrees / quantiles['0.975'][0])),
            float(deviation * mathematics.sqrt(degrees / quantiles['0.025'][0]))],
        'standardDeviationUpper95Ms': float(deviation * mathematics.sqrt(degrees / quantiles['0.05'][0]))})
fixture = {'syntheticOnly': True, 'blockMeans': blocks, 'expectedContrasts': expected,
    'quantiles': {probability: {'value80Digits': str(value), 'cdfResidual': str(residual)} for probability, (value, residual) in quantiles.items()},
    'decimalPrecision': 80, 'generator': 'mpmath==1.3.0', 'performanceAccepted': None, 'gateReevaluated': False}
(directory / 'variance-reference.json').write_bytes(json.dumps(fixture, indent=2).encode() + b'\n')
output = repository / 'reports/validation/jsdom-block-pilot/variance/20261026'
output.mkdir(parents=True, exist_ok=False)
(output / 'reference.json').write_bytes(json.dumps(fixture, indent=2).encode() + b'\n')
(output / 'generator.py.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps({'quantiles': policy, 'maximumCdfResidual': str(max(abs(value[1]) for value in quantiles.values())),
    'referenceSha256': hashlib.sha256((output / 'reference.json').read_bytes()).hexdigest()}))
