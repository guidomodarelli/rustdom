"""Transport/authenticate one functional CI; archives are data, never executable input."""
import sys
if sys.flags.optimize:
    raise RuntimeError('Functional hosted transport requires non-optimized Python; unset PYTHONOPTIMIZE')
import datetime
import hashlib
import io
import json
from pathlib import Path, PurePosixPath
import stat
import subprocess
import zipfile

repository = Path(__file__).resolve().parent.parent
primary = Path(r'C:\Users\guido\ghq\projects\rustdom')
root = repository / '.cache/pilot-functional-hosted-40301728367'
root.mkdir(exist_ok=True)
prefix = 'repos/guidomodarelli/rustdom/'
commit = '2466b8d35910e5f663224203462d341fdae14f64'

def api(endpoint):
    return subprocess.check_output(['gh', 'api', prefix + endpoint], cwd=repository, timeout=120)

def preserve(filename, raw):
    path = root / filename
    if path.exists():
        assert path.read_bytes() == raw, 'Preserved transport changed: ' + filename
    else:
        with path.open('xb') as stream: stream.write(raw)

run_raw = api('actions/runs/40301728367/attempts/1'); run = json.loads(run_raw)
jobs_raw = api('actions/runs/40301728367/attempts/1/jobs?per_page=100'); jobs = json.loads(jobs_raw)
artifacts_raw = api('actions/runs/40301728367/artifacts?per_page=100'); artifacts = json.loads(artifacts_raw)
assert run['head_sha'] == commit and run['run_attempt'] == 1 and run['event'] == 'pull_request'
assert run['path'] == '.github/workflows/jsdom-block-pilot-contracts.yml'
assert jobs['total_count'] == 2
observation = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S')
for label, raw in [('run', run_raw), ('jobs', jobs_raw), ('artifacts', artifacts_raw)]:
    preserve(observation + '-' + label + '.json', raw)
records = []
for job in jobs['jobs']:
    if job['status'] != 'completed': continue
    assert job['head_sha'] == commit and job['conclusion'] == 'success'
    version = next(version for version in ['22.12.0', '24.14.1'] if version in job['name'])
    name = 'jsdom-block-calendar-' + version + '-' + commit
    matches = [artifact for artifact in artifacts['artifacts'] if artifact['name'] == name]
    assert len(matches) == 1
    artifact = matches[0]
    assert artifact['workflow_run']['id'] == run['id'] and artifact['workflow_run']['head_sha'] == commit
    assert not artifact['expired'] and 0 < artifact['size_in_bytes'] <= 33554432
    assert job['started_at'] <= artifact['created_at'] <= job['completed_at']
    metadata = api('actions/artifacts/' + str(artifact['id'])); assert json.loads(metadata) == artifact
    preserve(version + '-metadata.json', metadata)
    filename = version + '-artifact.zip'
    raw = (root / filename).read_bytes() if (root / filename).exists() else api('actions/artifacts/' + str(artifact['id']) + '/zip')
    checksum = hashlib.sha256(raw).hexdigest()
    assert len(raw) == artifact['size_in_bytes'] and artifact['digest'] == 'sha256:' + checksum
    preserve(filename, raw)
    contents = {}; inventory = {}
    with zipfile.ZipFile(io.BytesIO(raw)) as archive:
        entries = archive.infolist(); names = [member.filename for member in entries]
        assert len(names) == len(set(names)) and len(entries) <= 1024
        assert sum(member.file_size for member in entries) <= 268435456
        for member in entries:
            path = PurePosixPath(member.filename)
            assert not path.is_absolute() and '..' not in path.parts and '\\' not in member.filename
            assert not stat.S_ISLNK(member.external_attr >> 16)
            if member.is_dir(): continue
            assert member.file_size <= 16777216
            data = archive.read(member)
            inventory[member.filename] = {'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}
            contents[member.filename] = data
    index = json.loads(contents['functional/index.json'])
    assert index['sourceIdentity']['sourceCommit'] == commit
    assert index['sourceIdentity']['sourceChangesRaw'] == '' and index['sourceIdentity']['sourceGitErrors'] == []
    for relative, expected in index['ownSourceFiles'].items():
        source = subprocess.check_output(['git', '-c', 'core.longpaths=true', 'show', commit + ':reports/benchmarks/native-class-ownership/control-precision/pilot/' + relative], cwd=repository)
        assert hashlib.sha256(source).hexdigest() == expected
    packet = {'index': index, 'workers': [
        {'report': json.loads(contents['functional/' + entry['directory'] + '/report.json']),
         'context': json.loads(contents['functional/' + entry['directory'] + '/report.json.context.json'])}
        for entry in index['processes']],
        'partial': json.loads(contents['functional/adverse/progress-failure/report.json'])}
    replay = []
    for reader_version, expected_hash in [
        ('22.12.0', 'b3b117a08ee61efee09e6fd523ab33c0c018da1b570bde08e4fd914dc1170ed6'),
        ('24.14.1', '58e74bf02fc5bbacc41dcb8bef089961cd5bddd37830b87784e4fc624d145d1f')]:
        executable = primary / ('.tools/node-win/node-v' + reader_version + '-win-x64/node.exe')
        assert hashlib.sha256(executable.read_bytes()).hexdigest() == expected_hash
        result = subprocess.run([str(executable), str(repository / '.cache/replay-pilot-functional-hosted.cjs')],
            input=json.dumps(packet).encode(), capture_output=True, cwd=repository, timeout=120)
        label = version + '-replay-' + reader_version
        preserve(label + '.stdout.json', result.stdout); preserve(label + '.stderr.log', result.stderr)
        assert result.returncode == 0, result.stderr.decode(errors='replace')
        result_record = json.loads(result.stdout); assert result_record['readerNode'] == 'v' + reader_version
        replay.append(result_record)
    assert replay[0]['totals'] == replay[1]['totals']
    preserve(version + '-members.json', json.dumps(inventory, indent=2).encode() + b'\n')
    logfile = version + '-job.log'
    if not (root / logfile).exists(): preserve(logfile, api('actions/jobs/' + str(job['id']) + '/logs'))
    records.append({'node': version, 'jobId': job['id'], 'artifactId': artifact['id'], 'bytes': len(raw), 'sha256': checksum,
        'sourceCommit': commit, 'sourceChangesRaw': index['sourceIdentity']['sourceChangesRaw'],
        'sourceGitErrors': index['sourceIdentity']['sourceGitErrors'], 'members': len(inventory), 'offlineReplays': replay})
receipt = {'sourceCommit': commit, 'runId': run['id'], 'attempt': 1, 'status': run['status'], 'conclusion': run['conclusion'],
    'jobsPreserved': len(records), 'complete': run['status'] == 'completed' and run['conclusion'] == 'success' and len(records) == 2,
    'records': records, 'archivedCodeExecuted': False, 'scientificCaptureExecuted': False, 'performanceAccepted': None, 'gateReevaluated': False}
preserve(observation + '-receipt.json', json.dumps(receipt, indent=2).encode() + b'\n')
print(json.dumps(receipt))
