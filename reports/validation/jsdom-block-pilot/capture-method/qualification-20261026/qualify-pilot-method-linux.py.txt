commands = []

def run(label, argv, environment=None, timeout=960):
    started = time.monotonic()
    record = {'label': label, 'argv': list(map(str, argv)), 'sourceCommit': head,
              'sourceIsPublishedCommit': False, 'started': True, 'scientificCaptureExecuted': False}
    commands.append(record)
    (output / 'commands.json').write_bytes(json.dumps(commands, indent=2).encode() + b'\n')
    with (output / (label + '.stdout.log')).open('xb') as stdout, (output / (label + '.stderr.log')).open('xb') as stderr:
        child = subprocess.Popen(record['argv'], cwd=source, env=environment, stdout=stdout, stderr=stderr)
        record['pid'] = child.pid
        print(json.dumps({'label': label, 'pid': child.pid, 'started': True}), flush=True)
        code = child.wait(timeout=timeout)
    record.update(exitCode=code, elapsedSeconds=time.monotonic() - started)
    (output / 'commands.json').write_bytes(json.dumps(commands, indent=2).encode() + b'\n')
    assert code == 0, label + ': ' + (output / (label + '.stderr.log')).read_text(errors='replace')
    print(json.dumps({'label': label, 'exitCode': code, 'elapsedSeconds': record['elapsedSeconds']}), flush=True)
    return (output / (label + '.stdout.log')).read_bytes()
