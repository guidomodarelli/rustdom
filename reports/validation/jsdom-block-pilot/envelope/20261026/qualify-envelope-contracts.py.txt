import sys
if sys.flags.optimize:
    raise RuntimeError('Contract qualification requires non-optimized Python')
import hashlib
import json
from pathlib import Path
import subprocess

repository = Path(__file__).resolve().parent.parent
primary = Path(r'C:\Users\guido\ghq\projects\rustdom')
root = repository / 'reports/validation/jsdom-block-pilot/envelope/20261026/contracts'
root.mkdir(parents=True, exist_ok=False)
pilot = 'reports/benchmarks/native-class-ownership/control-precision/pilot/'
tests = [pilot + name for name in ['blocks.spec.cjs', 'variance.spec.cjs', 'process-reader.spec.cjs', 'worker.spec.cjs', 'envelope.spec.cjs']]
records = []
for platform, version, relative, expected in [
    ('win32', '22.12.0', '.tools/node-win/node-v22.12.0-win-x64/node.exe', 'b3b117a08ee61efee09e6fd523ab33c0c018da1b570bde08e4fd914dc1170ed6'),
    ('win32', '24.14.1', '.tools/node-win/node-v24.14.1-win-x64/node.exe', '58e74bf02fc5bbacc41dcb8bef089961cd5bddd37830b87784e4fc624d145d1f'),
    ('linux', '22.12.0', '.tools/node-22/bin/node', '177208bfc4a9403121a40c72d038c670f4fd937fa16ca7df0a720e90be0fe2d9'),
    ('linux', '24.14.1', '.tools/node/bin/node', 'bf9112eb83a827dc9f9c8d19bb7bddcbf00f7b2fedae18fdcaee05406981f20a')]:
    executable = primary / relative
    assert hashlib.sha256(executable.read_bytes()).hexdigest() == expected
    if platform == 'win32': prefix = [str(executable)]; cwd = repository
    else:
        mounted = '/mnt/c/Users/guido/ghq/projects/rustdom/' + relative
        prefix = ['wsl', '-d', 'Rustdom-Validation-20261025', '-u', 'root', '--cd', '/dev/shm/rustdom-envelope-source-20261026', '--', mounted]
        cwd = repository
    assert subprocess.check_output([*prefix, '--version'], cwd=cwd).decode().strip() == 'v' + version
    result = subprocess.run([*prefix, '--test', *tests], cwd=cwd, capture_output=True, timeout=120)
    label = platform + '-' + version
    (root / (label + '.stdout.log')).write_bytes(result.stdout); (root / (label + '.stderr.log')).write_bytes(result.stderr)
    records.append({'platform': platform, 'node': version, 'executableSha256': expected, 'exitCode': result.returncode})
    assert result.returncode == 0, result.stderr.decode(errors='replace') + result.stdout.decode(errors='replace')
lint = subprocess.run(['wsl', '-d', 'Rustdom-Validation-20261025', '-u', 'root', '--cd',
    '/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom', '--',
    '/mnt/c/Users/guido/ghq/projects/rustdom/.tools/actionlint/actionlint', '.github/workflows/jsdom-block-pilot-contracts.yml'],
    capture_output=True, timeout=60)
(root / 'actionlint.stdout.log').write_bytes(lint.stdout); (root / 'actionlint.stderr.log').write_bytes(lint.stderr)
assert lint.returncode == 0
record = {'sdkPlatforms': records, 'testsPerSdkPlatform': 31, 'testsTotal': 124, 'newTestsPerSdkPlatform': 9,
    'historicalFunctionalReportsPerSdkPlatform': 32, 'historicalDataPromotedToScientific': False,
    'actionlintLinuxExitCode': lint.returncode, 'domExecuted': False,
    'scientificCaptureExecuted': False, 'performanceAccepted': None, 'gateReevaluated': False}
(root / 'summary.json').write_bytes(json.dumps(record, indent=2).encode() + b'\n')
print(json.dumps(record))
