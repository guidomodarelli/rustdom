import sys
if sys.flags.optimize:
    raise RuntimeError('Host-context recording requires non-optimized Python')
import datetime
import json
import os
from pathlib import Path
import platform

cpu = Path('/proc/cpuinfo').read_text()
model = next(line.split(':', 1)[1].strip() for line in cpu.splitlines() if line.startswith('model name'))
output = Path('/mnt/c/Users/guido/.codex/worktrees/native-class-ownership/rustdom/reports/validation/jsdom-block-pilot/envelope/20261026/host-context.json')
assert not output.exists()
record = {'observedAtUTC': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'observationStage': 'after-functional-qualification', 'system': platform.system(),
    'release': platform.release(), 'machine': platform.machine(), 'cpuModel': model,
    'logicalCpus': os.cpu_count(), 'observerAllowedCpus': sorted(os.sched_getaffinity(0)),
    'workerAffinities': 'Actual before/after masks retained separately in each raw worker report',
    'rootMountProbe': '/dev/sdd rw observed after normal restart; no repair command executed',
    'peakMemoryMeasured': False, 'scientificCaptureExecuted': False}
output.write_bytes(json.dumps(record, indent=2).encode() + b'\n')
print(json.dumps({key: record[key] for key in ['system', 'release', 'machine', 'logicalCpus']}))
