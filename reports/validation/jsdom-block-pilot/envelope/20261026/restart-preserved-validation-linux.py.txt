"""Restart only the owned failed validation distro after verifying every readable volatile file byte is retained."""
import sys
if sys.flags.optimize:
    raise RuntimeError('Validation recovery requires non-optimized checks')
import hashlib
import json
from pathlib import Path
import subprocess

repository = Path(__file__).resolve().parent.parent
cache = repository / '.cache'
root = cache / 'linux-recovery-envelope-20261026'
root.mkdir(exist_ok=False)
backup = json.loads((cache / 'pilot-functional-ram-preserved-20261026.json').read_bytes())
assert backup['allRegularBytesVerified'] and backup['publishedFunctionalEvidenceCommit'] == '9a6f62a982c64f5c02c54c6cd80b5a4d8474b6b7'
with (cache / backup['archive']).open('rb') as stream:
    assert hashlib.file_digest(stream, 'sha256').hexdigest() == backup['sha256']
records = []

def run(label, command):
    result = subprocess.run(command, capture_output=True, timeout=60)
    (root / (label + '.stdout.log')).write_bytes(result.stdout)
    (root / (label + '.stderr.log')).write_bytes(result.stderr)
    records.append({'label': label, 'command': command, 'exitCode': result.returncode})
    (root / 'commands.json').write_bytes(json.dumps(records, indent=2).encode() + b'\n')
    print(json.dumps(records[-1]), flush=True)
    return result

def running_names(raw):
    text = raw.decode('utf-16le') if b'\0' in raw else raw.decode('utf-8-sig')
    return [name.strip() for name in text.splitlines() if name.strip()]

result = run('running-before', ['wsl', '--list', '--running', '--quiet'])
assert result.returncode == 0 and running_names(result.stdout) == ['Rustdom-Validation-20261025']
assert run('terminate-owned', ['wsl', '--terminate', 'Rustdom-Validation-20261025']).returncode == 0
result = run('running-after-terminate', ['wsl', '--list', '--running', '--quiet'])
assert result.returncode == 0 and running_names(result.stdout) == []
assert run('shutdown-no-user-distro', ['wsl', '--shutdown']).returncode == 0
result = run('probe-owned-root', ['wsl', '-d', 'Rustdom-Validation-20261025', '-u', 'root', '--exec', '/usr/bin/id'])
receipt = {'distro': 'Rustdom-Validation-20261025', 'rootProbeSucceeded': result.returncode == 0,
    'volatileRegularBytesPreserved': backup['rawRegularBytes'], 'backupArchiveSha256': backup['sha256'],
    'volatileMetadataComplete': backup['metadataComplete'], 'unresolvedTemporaryLinks': backup['unresolvedReparseTargets'],
    'linksIndependentlyPreservedInSealedPackage': 'reports/validation/range-numeric-composed-package/20261026/archive.json',
    'otherRunningDistroTouched': False, 'scientificCaptureRepeated': False, 'filesystemRepairExecuted': False}
(root / 'restart.json').write_bytes(json.dumps(receipt, indent=2).encode() + b'\n')
print(json.dumps(receipt))
