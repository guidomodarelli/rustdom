"""Preserve authenticated functional CI artifacts; data transport only, never executes archived code."""
from pathlib import Path, PurePosixPath
import hashlib
from datetime import datetime
import json
import os
import stat
import subprocess
import unicodedata
import zipfile

repository = Path(__file__).resolve().parent.parent
head = '7148577dcf1eeffda2ebf95a3ccd1eb173a6082b'
run_id = 40344594967
prefix = 'repos/guidomodarelli/rustdom/'
root = repository / '.cache/pilot-method-hosted-40344594967'

def api(endpoint):
    return subprocess.check_output(['gh','api',prefix+endpoint], timeout=120)

def timestamp(value):
    return datetime.fromisoformat(value.replace('Z','+00:00'))

def digest(file):
    checksum=hashlib.sha256()
    with file.open('rb') as stream:
        for chunk in iter(lambda:stream.read(1048576),b''):checksum.update(chunk)
    return checksum.hexdigest()

run_raw = api(f'actions/runs/{run_id}/attempts/1')
run = json.loads(run_raw)
assert run['head_sha']==head and run['run_attempt']==1
assert run['path']=='.github/workflows/jsdom-block-pilot-contracts.yml'
assert run['status']=='completed', 'Observe the same run; do not restart it or collect a live artifact'
root.mkdir(exist_ok=False)
(root/'run.json').write_bytes(run_raw)
jobs_raw = api(f'actions/runs/{run_id}/attempts/1/jobs?per_page=100')
artifacts_raw = api(f'actions/runs/{run_id}/artifacts?per_page=100')
(root/'jobs.json').write_bytes(jobs_raw); (root/'artifacts.json').write_bytes(artifacts_raw)
jobs=json.loads(jobs_raw); artifacts=json.loads(artifacts_raw)
assert jobs['total_count']==len(jobs['jobs']) <= 4
assert artifacts['total_count']==len(artifacts['artifacts']) <= 4
for job in jobs['jobs']:
    assert job['head_sha']==head and job['run_attempt']==1
    if job['started_at'] is not None and job['conclusion']!='skipped':
        (root/(str(job['id'])+'.log')).write_bytes(api(f'actions/jobs/{job["id"]}/logs'))
allowed={f'jsdom-block-{role}-{version}-{head}' for role in ['calendar','replay'] for version in ['22.12.0','24.14.1']}
seen=set(); records=[]
for artifact in artifacts['artifacts']:
    assert artifact['name'] in allowed and artifact['name'] not in seen
    seen.add(artifact['name'])
    assert artifact['workflow_run']['id']==run_id and artifact['workflow_run']['head_sha']==head
    assert not artifact['expired'] and 0<artifact['size_in_bytes']<=536870912
    role,version=next((role,version) for role in ['calendar','replay'] for version in ['22.12.0','24.14.1']
                      if artifact['name']==f'jsdom-block-{role}-{version}-{head}')
    job_name=('Calendario y medias Node ' if role=='calendar' else 'Replay transportado Node ')+version
    matching_jobs=[job for job in jobs['jobs'] if job['name']==job_name]
    assert len(matching_jobs)==1
    job=matching_jobs[0]
    assert timestamp(job['started_at'])<=timestamp(artifact['created_at'])<=timestamp(job['completed_at'])
    metadata=api(f'actions/artifacts/{artifact["id"]}'); assert json.loads(metadata)==artifact
    (root/(str(artifact['id'])+'-metadata.json')).write_bytes(metadata)
    archive_path=root/(str(artifact['id'])+'.zip')
    with archive_path.open('xb') as stream:
        subprocess.run(['gh','api',prefix+f'actions/artifacts/{artifact["id"]}/zip'],stdout=stream,timeout=240,check=True)
    assert archive_path.stat().st_size==artifact['size_in_bytes']
    checksum=digest(archive_path)
    assert artifact['digest']=='sha256:'+checksum
    output=root/artifact['name']; output.mkdir()
    inventory=[]; portable_names=set(); total=0
    with zipfile.ZipFile(archive_path) as archive:
        assert len(archive.infolist())<=20000
        for member in archive.infolist():
            path=PurePosixPath(member.filename)
            assert member.filename==path.as_posix()+('/' if member.is_dir() else '') and not path.is_absolute()
            assert unicodedata.normalize('NFC',member.filename)==member.filename
            assert all(part not in ['.','..',''] and ':' not in part and '\\' not in part for part in path.parts)
            assert len(member.filename.encode())<=255 and not member.flag_bits&1
            kind=stat.S_IFMT(member.external_attr>>16)
            assert kind in [0,stat.S_IFREG,stat.S_IFDIR] and not stat.S_ISLNK(member.external_attr>>16)
            if member.is_dir():
                continue
            portable=member.filename.casefold(); assert portable not in portable_names
            portable_names.add(portable)
            assert 0<=member.file_size<=268435456
            total+=member.file_size; assert total<=2147483648
            target=output.joinpath(*path.parts)
            target.parent.mkdir(parents=True,exist_ok=True)
            checksum_member=hashlib.sha256(); count=0
            with archive.open(member) as source, target.open('xb') as destination:
                for chunk in iter(lambda:source.read(1048576),b''):
                    count+=len(chunk); assert count<=member.file_size
                    checksum_member.update(chunk); destination.write(chunk)
            assert count==member.file_size
            inventory.append({'path':member.filename,'bytes':count,'crc32':member.CRC,'sha256':checksum_member.hexdigest()})
    records.append({'id':artifact['id'],'name':artifact['name'],'bytes':archive_path.stat().st_size,
                    'sha256':checksum,'members':inventory})
receipt={'headCommit':head,'runId':run_id,'attempt':1,'conclusion':run['conclusion'],
         'allFourJobsPassed':jobs['total_count']==4 and all(job['conclusion']=='success' for job in jobs['jobs']),
         'allFourArtifactsPresent':seen==allowed,'artifacts':records,'DOMExecutedByCollector':False,
         'offlineReplayPerformed':False,'performanceAccepted':None,'gateReevaluated':False}
(root/'receipt.json').write_bytes(json.dumps(receipt,indent=2).encode()+b'\n')
print(json.dumps({key:value for key,value in receipt.items() if key!='artifacts'}))
