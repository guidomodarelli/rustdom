def digest(filename):
    checksum = hashlib.sha256()
    with filename.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1048576), b''):
            checksum.update(chunk)
    return checksum.hexdigest()

def read(filename):
    return json.loads(filename.read_bytes())

profile = read(root / 'source-profile-authentication.json')
for section in ['frozenFiles', 'additionalFrozenFiles']:
    for relative, expected in profile[section].items():
        assert digest(root / 'source' / relative) == expected, relative

files = {}

def select(location, prefix):
    assert location.exists(), str(location)
    members = [location] if location.is_file() else sorted(location.rglob('*'))
    for member in members:
        assert not member.is_symlink(), str(member)
        if member.is_file():
            relative = prefix if location.is_file() else prefix + '/' + member.relative_to(location).as_posix()
            assert relative not in files
            files[relative] = member

for name in ['node22', 'node24']:
    cohort = root / (name + '-authentication')
    index = read(cohort / 'index.json')
    assert index['complete'] and index['functionalOnly'] and not index['scientificCaptureExecuted']
    assert index['performanceAccepted'] is None and not index['gateReevaluated']
    assert not pathlib.Path('/proc/' + str(index['parentPid'])).exists()
    assert len(index['processes']) == 64 and len(index['adverse']) == 6 and len(index['parentMemory']) == 3

        assert digest(current / method / relative) == expected
    totals = dict(processes=0, measured=0, warmup=0, windows=0, documents=0,
                  firstUnclearStates=0, finalClearStates=0, nativeWorkers=0)
    endpoints = []
    for entry in index['processes']:
        command = entry['invocation']['command']
        assert entry['complete'] and command['complete'] and command['exitCode'] == 0
        assert not command['timedOut'] and not command['cancelled'] and not command.get('leftoverProcessGroupDetected', False)
        assert not pathlib.Path('/proc/' + str(command['pid'])).exists()
        report = read(cohort / entry['directory'] / 'report.json')
        assert report['complete'] and report['functionalOnly'] and not report['performanceStudyExecuted']
        assert len(report['measurement']['measuredSamplesMs']) == 144 and len(report['measurement']['warmupSamplesMs']) == 48
        assert report['trackedOwners'] == dict(window=12, document=12)
        assert len(report['cleanupTrace']) == 3 and all(state['clear'] for state in report['cleanupTrace'][-2:])
        totals['processes'] += 1
        totals['measured'] += 144
        totals['warmup'] += 48
        totals['windows'] += 12
        totals['documents'] += 12
        totals['firstUnclearStates'] += int(not report['cleanupTrace'][0]['clear'])
        totals['finalClearStates'] += 2
        if report['engine'] == 'rustdom':
            assert report['nativeBaseline'] is not None
            totals['nativeWorkers'] += 1
        else:
            assert report['nativeBaseline'] is None
        endpoints.append(dict(input=entry['input'], nativeBaseline=report['nativeBaseline'], cleanupTrace=report['cleanupTrace']))
    for entry in index['adverse']:
        command = entry['command']
        assert command['complete'] and command['exitCode'] == 2 and not command['timedOut'] and not command['cancelled']
        assert not command.get('leftoverProcessGroupDetected', False)
        assert not pathlib.Path('/proc/' + str(command['pid'])).exists()
    failure = read(cohort / 'adverse/progress-failure/report.json')
    assert not failure['complete'] and failure['trackedOwners'] == dict(window=1, document=1)
    assert all(state['clear'] for state in failure['cleanupTrace'][-2:])
    tests = root / (name + '-authentication-tests-final')
    receipt = read(tests / 'receipts.json')
    assert len(receipt['commands']) == 5 and all(command['exitCode'] == 0 for command in receipt['commands'])
    production = read(tests / 'production.stdout.log')
    assert len(production['observedInputs']) == 344 and len(production['differences']) == 4
    assert not production['matchesSelectedCompilerInputs'] and not production['unregisteredBuildInputs']
    summary['cohorts'].append(dict(node=index['node'], totals=totals, elapsedMs=index['elapsedMs'],
                                   parentGone=True, workerCommandsClosed=70, sourceFiles=index['ownSourceFiles']))
    resources['cohorts'].append(dict(node=index['node'], totals=totals, parentMemory=index['parentMemory'], workers=endpoints,
                                    deliberateFailureCleanup=failure['cleanupTrace']))
    select(cohort, cohort.name)
    select(tests, tests.name)

with tarfile.open(archive, 'r:gz') as bundle:
    members = bundle.getmembers()
    assert len(members) == len(files)
    for member in members:
        assert member.isfile()
        raw = bundle.extractfile(member).read()
        if member.name != 'manifest.json':
            assert manifest[member.name] == dict(bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
