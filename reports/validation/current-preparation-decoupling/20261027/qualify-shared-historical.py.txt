extraction=output/'preparation'
with tarfile.open(archive,'r:gz') as bundle:
    members=bundle.getmembers(); assert sum(member.size for member in members)<1024*1024*1024
    assert len(set(member.name.rstrip('/') for member in members))==len(members)
    for member in members:
        name=pathlib.PurePosixPath(member.name)
        assert not name.is_absolute() and '..' not in name.parts and name.parts[0]=='ownership-build'
        destination=extraction/member.name;assert destination.resolve().is_relative_to(extraction)
        assert member.isdir() or member.isfile() or member.issym()
        if member.isdir(): destination.mkdir(parents=True,exist_ok=True)
        elif member.isfile():
            destination.parent.mkdir(parents=True,exist_ok=True)
            with destination.open('xb') as stream: shutil.copyfileobj(bundle.extractfile(member),stream)
            destination.chmod(0o700 if member.mode & 0o100 else 0o600)
    for member in members:
        if not member.issym():continue
        destination=extraction/member.name;target=pathlib.Path(member.linkname)
        assert not target.is_absolute() and (destination.parent/target).resolve().is_relative_to(extraction/'ownership-build')
        assert not destination.exists() and not destination.is_symlink()
        destination.parent.mkdir(parents=True,exist_ok=True);destination.symlink_to(member.linkname)
