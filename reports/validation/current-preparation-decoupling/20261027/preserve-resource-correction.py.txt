def copy(source,target):
    target.parent.mkdir(parents=True,exist_ok=True);assert not target.exists()
    shutil.copyfile(source,target);assert hashlib.sha256(source.read_bytes()).digest()==hashlib.sha256(target.read_bytes()).digest()

        copy(member,output/'shared'/source.name/member.name)
    report=source/'functional.json';receipt=json.loads((source/'functional.process.json').read_bytes())
    assert hashlib.sha256(report.read_bytes()).hexdigest()==receipt['functionalReportSha256']
    assert len(json.loads(report.read_bytes())['cases'])==38
    memory=root/('reports/memory/current-preparation-decoupling/20261027/'+label+'-functional.json')
