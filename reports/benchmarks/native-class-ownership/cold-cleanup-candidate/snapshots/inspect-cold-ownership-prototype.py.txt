"""Inspect generated code as data; neither shared library is loaded or benchmarked."""
import hashlib
import json
from pathlib import Path
import subprocess

root = Path('/home/guido/rustdom-validation/ownership-cold-cleanup-prototype')
inspection = {'originCommit': 'f44f5dd0f9317cfaa24bc9e7e59574fad632ac98',
    'method': 'Same source path, compiler1.99.1 and release flags; only PendingClassValue::drop cold/noinline attributes change',
    'benchmarkExecuted': False, 'performanceAccepted': None, 'timingConclusion': None, 'variants': {}}
for variant in ['baseline', 'cold-cleanup']:
    binary = root / 'artifacts' / (variant + '.node')
    rows = []
    for line in (root / 'logs' / (variant + '-symbols.stdout.log')).read_text().splitlines():
        fields = line.split(maxsplit=3)
        if len(fields) != 4:
            continue
        address, size, kind, name = fields
        if ('NativeMutationRecord' in name and 'Option<' not in name and
            (name.endswith('::to_napi_value') or 'PendingClassValue' in name)) or name.endswith('::class::new_instance_raw'):
            rows.append({'address': int(address, 16), 'bytes': int(size, 16), 'kind': kind, 'symbol': name})
    assert sum('NativeMutationRecord' in row['symbol'] and row['symbol'].endswith('::to_napi_value') for row in rows) == 1
    wrappers = [line for line in (root / 'logs' / (variant + '-symbols.stdout.log')).read_text().splitlines()
        if '::class::new_instance::<' in line and 'NativeMutationRecord' in line]
    for ordinal, row in enumerate(rows):
        filename = root / 'artifacts' / f'{variant}-{ordinal}.asm'
        content = subprocess.check_output(['objdump', '-d', '-C', '--no-show-raw-insn',
            '--start-address=' + hex(row['address']), '--stop-address=' + hex(row['address'] + row['bytes']), binary])
        filename.write_bytes(content)
        row['disassembly'] = filename.name
        row['disassemblySha256'] = hashlib.sha256(content).hexdigest()
    inspection['variants'][variant] = {'binarySha256': hashlib.sha256(binary.read_bytes()).hexdigest(),
        'binaryBytes': binary.stat().st_size, 'selectedFunctions': rows, 'typedNewInstanceWrappers': wrappers}
for tool in ['nm', 'objdump', 'readelf']:
    inspection.setdefault('tools', {})[tool] = subprocess.check_output([tool, '--version'], text=True).splitlines()[0]
(root / 'inspection.json').write_text(json.dumps(inspection, indent=2) + '\n')
print(json.dumps({'functions': {name: [row['bytes'] for row in record['selectedFunctions'] if row['symbol'].endswith('::to_napi_value')]
    for name, record in inspection['variants'].items()}, 'benchmarkExecuted': False, 'performanceAccepted': None}))
